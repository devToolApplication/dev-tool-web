# SDK Task Console Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking. When dispatching subagents, use only the dedicated configured agents: `dev-be-agent` for backend tasks, `dev-fe-agent` for frontend tasks, and `test-qa-agent` for verification.

**Goal:** Build the `/admin/system-management/ai-agent-execution` SDK task console with full prompt run input, MCRS-backed server history, selected run detail, and polling for active runs.

**Architecture:** Extend the SDK task API contract first so the full run form is valid and history can show a sanitized request snapshot. Add a thin AI Agent MCRS admin proxy under `/v1/admin/sdk/tasks` that wraps SDK payloads in `BaseResponse.data`. Add a focused Angular `system-management` feature with local page state, one HTTP service, and existing shared UI primitives.

**Tech Stack:** Node 24 + Express + Zod + tsoa, Java 21 + Spring Boot + OpenFeign, Angular 21 + signals + shared `app-*` controls + Playwright.

```ts
export type AiTaskRunRequestSnapshot = {
  agentCode: string;
  provider?: AgentProvider;
  prompt: string;
  workingDirectory?: string;
  threadId?: string;
  model?: string;
  reasoningEffort?: string;
  outputSchema?: JsonObject;
  requestContext?: JsonObject;
  callbackUrl?: string;
  callbackAuthSecretConfigured?: boolean;
};

export type AiTaskRunDetail = AiTaskRunSummary & {
  request?: AiTaskRunRequestSnapshot;
  events: AiTaskRunEvent[];
  result?: AiTaskExecuteResult;
};

export type AiTaskRunCreate = AiTaskRunSummary & {
  request?: AiTaskRunRequestSnapshot;
};

export function toAiTaskRunCreate(
  taskId: string,
  input: AiTaskExecuteInput,
  defaultWorkingDirectory: string,
  now: string
): AiTaskRunCreate {
  const workingDirectory = input.workingDirectory || defaultWorkingDirectory;
  return {
    taskId,
    status: "RUNNING",
    agentCode: input.agentCode,
    provider: input.provider,
    workingDirectory,
    threadId: input.threadId,
    model: input.model,
    reasoningEffort: input.reasoningEffort,
    promptPreview: input.prompt.slice(0, 500),
    createdAt: now,
    updatedAt: now,
    request: {
      agentCode: input.agentCode,
      provider: input.provider,
      prompt: input.prompt,
      workingDirectory,
      threadId: input.threadId,
      model: input.model,
      reasoningEffort: input.reasoningEffort,
      outputSchema: input.outputSchema,
      requestContext: input.requestContext,
      callbackUrl: input.callbackUrl,
      callbackAuthSecretConfigured: !!input.callbackAuthSecretCode
    }
  };
}
```

```ts
import { HttpClient, HttpParams } from '@angular/common/http';
import { Injectable } from '@angular/core';
import { map, Observable } from 'rxjs';

import { BaseResponse } from '@core/http/base-response.model';
import { environment } from '../../../../../enviroment/environment';
import type {
  SdkTaskRunDetail,
  SdkTaskRunListQuery,
  SdkTaskRunListResponse,
  SdkTaskRunRequest,
  SdkTaskRunSummary,
} from '../model/sdk-task.model';

@Injectable({ providedIn: 'root' })
export class SdkTaskApiService {
  private readonly baseUrl = `${environment.apiUrl.adminAiGenerator}/sdk/tasks`;

  constructor(private readonly http: HttpClient) {}

  startRun(request: SdkTaskRunRequest): Observable<SdkTaskRunSummary> {
    return this.http
      .post<BaseResponse<SdkTaskRunSummary>>(`${this.baseUrl}/runs`, request)
      .pipe(map((response) => response.data));
  }

  listRuns(query: SdkTaskRunListQuery = {}): Observable<SdkTaskRunListResponse> {
    return this.http
      .get<BaseResponse<SdkTaskRunListResponse>>(`${this.baseUrl}/runs`, { params: this.queryParams(query) })
      .pipe(map((response) => response.data));
  }

  getRun(taskId: string): Observable<SdkTaskRunDetail> {
    return this.http
      .get<BaseResponse<SdkTaskRunDetail>>(`${this.baseUrl}/runs/${encodeURIComponent(taskId)}`)
      .pipe(map((response) => response.data));
  }

  private queryParams(query: SdkTaskRunListQuery): HttpParams {
    let params = new HttpParams();
    Object.entries(query).forEach(([key, value]) => {
      if (value !== undefined && value !== null && value !== '') {
        params = params.set(key, String(value));
      }
    });
    return params;
  }
}
```

```ts
import { Component, OnDestroy, OnInit, computed, inject, signal } from '@angular/core';
import { firstValueFrom } from 'rxjs';
import type { AppTabItem } from '@shared/ui/primitives/tabs/tabs.component';
import { SdkTaskApiService } from '../../api/sdk-task-api.service';
import type { SdkTaskProvider, SdkTaskRunDetail, SdkTaskRunListQuery, SdkTaskRunRequest, SdkTaskRunSummary, SdkTaskStatus } from '../../model/sdk-task.model';

interface SdkTaskConsoleForm {
  agentCode: string;
  provider: SdkTaskProvider | null;
  prompt: string;
  threadId: string;
  workingDirectory: string;
  model: string;
  reasoningEffort: string;
  outputSchemaText: string;
  requestContextText: string;
  callbackUrl: string;
  callbackAuthSecretCode: string;
}

const EMPTY_FORM: SdkTaskConsoleForm = {
  agentCode: '',
  provider: 'codex',
  prompt: '',
  threadId: '',
  workingDirectory: '',
  model: 'gpt-5.2',
  reasoningEffort: 'medium',
  outputSchemaText: '',
  requestContextText: '',
  callbackUrl: '',
  callbackAuthSecretCode: '',
};

const TERMINAL_STATUSES: SdkTaskStatus[] = ['BLOCKED_CONFIG', 'BLOCKED_MCP', 'COMPLETED', 'FAILED', 'FAILED_DEPENDENCY', 'TIMEOUT'];

@Component({
  selector: 'app-sdk-task-console',
  standalone: false,
  templateUrl: './sdk-task-console.component.html',
  styleUrl: './sdk-task-console.component.css',
})
export class SdkTaskConsoleComponent implements OnInit, OnDestroy {
  private readonly api = inject(SdkTaskApiService);
  private pollTimer: ReturnType<typeof setInterval> | null = null;

  readonly form = signal<SdkTaskConsoleForm>({ ...EMPTY_FORM });
  readonly runs = signal<SdkTaskRunSummary[]>([]);
  readonly selectedRunDetail = signal<SdkTaskRunDetail | null>(null);
  readonly selectedTab = signal('summary');
  readonly loadingRuns = signal(false);
  readonly loadingDetail = signal(false);
  readonly submitting = signal(false);
  readonly error = signal<string | null>(null);
  readonly formError = signal<string | null>(null);
  readonly query = signal<SdkTaskRunListQuery>({ page: 1, size: 20 });

  readonly tabs: AppTabItem[] = [
    { label: 'Summary', value: 'summary' },
    { label: 'Request', value: 'request' },
    { label: 'Events', value: 'events' },
    { label: 'Output', value: 'output' },
    { label: 'Raw JSON', value: 'raw' },
  ];
  readonly canRun = computed(() => !!this.form().agentCode.trim() && !!this.form().prompt.trim() && !this.submitting());

  ngOnInit(): void {
    void this.loadRuns();
  }

  ngOnDestroy(): void {
    this.stopPolling();
  }

  async loadRuns(): Promise<void> {
    this.loadingRuns.set(true);
    this.error.set(null);
    try {
      const response = await firstValueFrom(this.api.listRuns(this.query()));
      this.runs.set(response.items ?? []);
    } catch (error) {
      this.error.set(errorMessage(error));
    } finally {
      this.loadingRuns.set(false);
    }
  }

  async runPrompt(): Promise<void> {
    this.formError.set(null);
    const payload = this.buildRequest();
    if (!payload) return;
    this.submitting.set(true);
    try {
      const run = await firstValueFrom(this.api.startRun(payload));
      await this.loadRuns();
      this.selectRun(run);
    } catch (error) {
      this.formError.set(errorMessage(error));
    } finally {
      this.submitting.set(false);
    }
  }

  selectRun(run: SdkTaskRunSummary): void {
    this.stopPolling();
    void this.loadRunDetail(run.taskId);
  }

  async loadRunDetail(taskId: string): Promise<void> {
    this.loadingDetail.set(true);
    try {
      const detail = await firstValueFrom(this.api.getRun(taskId));
      this.selectedRunDetail.set(detail);
      if (detail.status === 'RUNNING') this.startPolling(taskId);
      if (TERMINAL_STATUSES.includes(detail.status)) this.stopPolling();
    } catch (error) {
      this.error.set(errorMessage(error));
    } finally {
      this.loadingDetail.set(false);
    }
  }

  resetForm(): void {
    this.form.set({ ...EMPTY_FORM });
    this.formError.set(null);
  }

  private startPolling(taskId: string): void {
    if (this.pollTimer) return;
    this.pollTimer = setInterval(() => void this.loadRunDetail(taskId), 5000);
  }

  private stopPolling(): void {
    if (!this.pollTimer) return;
    clearInterval(this.pollTimer);
    this.pollTimer = null;
  }

  private buildRequest(): SdkTaskRunRequest | null {
    const form = this.form();
    const agentCode = form.agentCode.trim();
    const prompt = form.prompt.trim();
    if (!agentCode || !prompt) {
      this.formError.set('systemManagement.sdkTask.validation.required');
      return null;
    }
    const outputSchema = parseJsonObject(form.outputSchemaText);
    const requestContext = parseJsonObject(form.requestContextText);
    if (outputSchema === false || requestContext === false) {
      this.formError.set('systemManagement.sdkTask.validation.jsonObject');
      return null;
    }
    return stripEmpty({
      agentCode,
      provider: form.provider ?? undefined,
      prompt,
      threadId: form.threadId.trim(),
      workingDirectory: form.workingDirectory.trim(),
      model: form.model.trim(),
      reasoningEffort: form.reasoningEffort.trim(),
      outputSchema,
      requestContext,
      callbackUrl: form.callbackUrl.trim(),
      callbackAuthSecretCode: form.callbackAuthSecretCode.trim(),
    });
  }
}

function parseJsonObject(text: string): Record<string, unknown> | undefined | false {
  const raw = text.trim();
  if (!raw) return undefined;
  try {
    const parsed = JSON.parse(raw) as unknown;
    return parsed && typeof parsed === 'object' && !Array.isArray(parsed) ? parsed as Record<string, unknown> : false;
  } catch {
    return false;
  }
}

function stripEmpty<T extends Record<string, unknown>>(value: T): T {
  return Object.fromEntries(Object.entries(value).filter(([, item]) => item !== undefined && item !== '')) as T;
}

function errorMessage(error: unknown): string {
  return error instanceof Error ? error.message : 'systemManagement.sdkTask.error.loadFailed';
}
```
